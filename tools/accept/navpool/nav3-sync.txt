// §152 navpool-nav3-sync: the replay of updateNav (stepDrop runs inside it) over a real page trace, for nav3 OR navpool — no Chrome.
// Copied from §145's tools/work/nav3/swell-replay.mjs (worktree agent-a43e535ac135b941f, md5 ab9f8ef77661d478ea453a03d1c9addf), with:
//   --scene nav3|navpool   the scene folder whose nav.js / implode.js is imported (one scene per process: module state is per path)
//   --music                presence := max(presence, MUSIC) on every frame, BEFORE the shift stage and updateNav (MUSIC hardcoded .3, the
//                          same for both scenes; for navpool the script asserts ROLL.IDLE + ROLL.IDLE_W === MUSIC once)
//   --dump <file>          one JSON line per frame: frame, t, NAV mode/c/ball/rho/th/rhoT/thT/swell/turn/cLo/cHi, IMP x/y/q/eps/drops/frames/T
//   --sum <file>           the run's events as JSON: shift frames, drop (implosion start) frames, re-arm frames, mode transitions [frame, from, to]
//   node tools/work/navpool/sync-replay.mjs <Track> --scene nav3|navpool [--music] [--dump f.jsonl] [--sum f.json] [--map1] [t1,t2,…]
import fs from 'node:fs';
const MUSIC = 0.3;
const W = new URL('../../../assets/', import.meta.url).pathname;
const a = process.argv.slice(2);
const opt = (k, d) => (a.includes(k) ? a.splice(a.indexOf(k), 2)[1] : d);
const flag = (k) => (a.includes(k) ? (a.splice(a.indexOf(k), 1), true) : false);
const SCENE = opt('--scene', 'nav3'), DUMP = opt('--dump', ''), SUM = opt('--sum', ''), MUS = flag('--music');
if (SCENE !== 'nav3' && SCENE !== 'navpool') throw new Error('--scene nav3|navpool');
const { NAV, updateNav } = await import(W + 'scenes/' + SCENE + '/nav.js');
const { IMP } = await import(W + 'scenes/' + SCENE + '/implode.js');
if (MUS && SCENE === 'navpool') {
  const { ROLL } = await import(W + 'scenes/navpool/nav.js');
  if (!(ROLL.IDLE + ROLL.IDLE_W === MUSIC)) throw new Error(`navpool ROLL.IDLE + ROLL.IDLE_W = ${ROLL.IDLE + ROLL.IDLE_W} !== ${MUSIC}`);
}
const { Shift, SHIFT } = await import(W + 'engine/shift/shift.js');
const { BUILD } = await import(W + 'engine/build/build.js');
const { baseLight } = await import(W + 'math/loudlight.js');
const MAP1 = flag('--map1'), track = a[0], times = (a[1] || '').split(',').filter(Boolean).map(Number);
const p = opt('--trace', new URL('../../truth/traces/' + track + (MAP1 ? '-map1' : '-map0') + '.json', import.meta.url).pathname);
const d = JSON.parse(fs.readFileSync(p)), F = d.fields, T = d.t, cols = d.cols, col = (j) => Array.isArray(cols) ? cols[j] : cols[F[j]];
const sg = cols.subGate ? cols.subGate.map(Number) : null, bl = cols.buildLive ? cols.buildLive.map(Number) : null;
const killed = new Set(), killedArm = new Set();
if (sg && bl) { let on = false, armOk = true; for (let i = 0; i < T.length; i++) { const v = bl[i] > 0.05; if (v && !on) { let s = 0, n = 0; for (let j = i; j >= 0 && T[i] - T[j] <= 32; j--) { s += sg[j]; n++; } armOk = n ? s / n >= BUILD.SUBV_HAD : true; } on = v; if (v && !armOk) killedArm.add(i); if (+cols.dropLiveEvt[i] > 0 && !armOk) killed.add(i); } }
const sh = new Shift(SHIFT), si = {}, S = {};
let prev = '', cur = null; const imps = [], ev = [], shifts = [], swellsAt = [], dump = [];
const E = { scene: SCENE, track, music: MUS, frames: T.length, t0: T[0], shifts: [], drops: [], rearms: [], trans: [] };
let sw0 = 0;
for (let i = 0; i < T.length; i++) {
  for (let j = 0; j < F.length; j++) S[F[j]] = col(j)[i];
  if (MUS) S.presence = Math.max(+S.presence, MUSIC);
  S.arc = 'idle';
  const dt = i ? T[i] - T[i - 1] : 1 / 60;
  if (killed.has(i)) S.dropLiveEvt = 0; if (killedArm.has(i)) S.buildLive = 0;
  si.B = +S.beatCount + +S.beatPhase; si.bpm = +S.bpm; si.ok = +S.presence >= SHIFT.PRESENT; si.dt = dt;
  si.lad = baseLight(+S.loudRel, +S.loudRange, +S.loudAbs, NaN); si.drop = S.dropLiveEvt || S.mapDropEvt ? 1 : 0; si.armed = +S.buildLive > 0 && !killedArm.has(i) ? 1 : 0;
  const o = sh.step(si); S.shiftEvt = o.shiftEvt; S.shiftHold = o.shiftHold;
  if (o.shiftEvt) { shifts.push(T[i].toFixed(1)); E.shifts.push(i); }
  const r0 = IMP.rearms || 0, d0 = IMP.drops || 0;
  updateNav(dt, T[i], S, { isLogical: true, drum: false, P: { reach: -6 } });
  if ((IMP.drops || 0) > d0) E.drops.push(i);
  if ((IMP.rearms || 0) > r0) E.rearms.push(i);
  if ((NAV.swells || 0) > sw0) { swellsAt.push(T[i].toFixed(1)); sw0 = NAV.swells; }
  if (NAV.mode !== prev) {
    ev.push(`${T[i].toFixed(1)} ${prev || '·'}→${NAV.mode}`); E.trans.push([i, prev || '·', NAV.mode]);
    if (NAV.mode === 'OUT' && prev !== 'IN') cur = { t0: T[i], rearms: 0 };
    if (NAV.mode === 'INT' && cur) { cur.t1 = T[i]; imps.push(cur); cur = null; }
    prev = NAV.mode;
  }
  if ((IMP.rearms || 0) > r0 && cur) cur.rearms++;
  if (DUMP) dump.push(JSON.stringify({ f: i, t: T[i], mode: NAV.mode, c: NAV.c, ball: NAV.ball, rho: NAV.rho, th: NAV.th, rhoT: [NAV.rhoT.x, NAV.rhoT.v], thT: [NAV.thT.x, NAV.thT.v], swell: NAV.swell, turn: NAV.turn, cLo: NAV.cLo, cHi: NAV.cHi, imp: { x: IMP.x, y: IMP.y, q: IMP.q, eps: IMP.eps, drops: IMP.drops, frames: IMP.frames, T: IMP.T } }));
  if (times.length && times.some((t) => T[i] >= t && (i === 0 || T[i - 1] < t))) console.log(`  t ${T[i].toFixed(2)} f${i} ${NAV.mode} ρ ${NAV.rho.toFixed(4)} θ ${NAV.th.toFixed(4)} turn ${NAV.turn.toFixed(4)} swell ${(NAV.swell || 0).toFixed(3)} drops ${IMP.drops}`);
}
if (cur) { cur.t1 = T[T.length - 1]; cur.open = 1; imps.push(cur); }
const f = (x, n = 1) => (+x).toFixed(n);
console.log(`${SCENE} ${track}${MAP1 ? ' [map1]' : ''}${MUS ? ' [music]' : ''}: ${T.length} frames ${f(T[0])}–${f(T[T.length - 1])} s | shifts ${shifts.join(' ') || '-'} | SWELLS ${NAV.swells || 0}: ${swellsAt.join(' ') || '-'} | implosions ${imps.length}: ${imps.map((m) => `${f(m.t0)}→${f(m.t1)}${m.open ? '+' : ''} (${m.rearms} re-arm)`).join(' | ')}`);
if (process.env.CHANGES) console.log('  ' + ev.join(' | '));
if (DUMP) fs.writeFileSync(DUMP, dump.join('\n') + '\n');
if (SUM) fs.writeFileSync(SUM, JSON.stringify(E));

## cmp6b.mjs
// §152 6b: nav3 vs navpool unclamped (raw) replays — triggers exact, transition sequence ±2 frames, implosion count, carry bounds
import fs from 'node:fs';
const dir = process.argv[2] || 'post';
const circ = (a, b) => { let d = (a - b) % 1; if (d > 0.5) d -= 1; if (d < -0.5) d += 1; return Math.abs(d); };
let allOk = true;
for (const tr of ['IBelongHere', 'TheChain', 'Comptine', 'SeeYouDrop']) {
  const s3 = JSON.parse(fs.readFileSync(`${dir}/${tr}-nav3-raw.sum.json`)), sp = JSON.parse(fs.readFileSync(`${dir}/${tr}-navpool-raw.sum.json`));
  const L3 = fs.readFileSync(`${dir}/${tr}-nav3-raw.jsonl`, 'utf8').trim().split('\n').map(JSON.parse), Lp = fs.readFileSync(`${dir}/${tr}-navpool-raw.jsonl`, 'utf8').trim().split('\n').map(JSON.parse);
  const eq = (a, b) => JSON.stringify(a) === JSON.stringify(b);
  const trig = eq(s3.shifts, sp.shifts) && eq(s3.drops, sp.drops) && eq(s3.rearms, sp.rearms);
  const seq = s3.trans.length === sp.trans.length && s3.trans.every((t, i) => t[1] === sp.trans[i][1] && t[2] === sp.trans[i][2] && Math.abs(t[0] - sp.trans[i][0]) <= 2);
  const maxDt = s3.trans.length === sp.trans.length ? Math.max(0, ...s3.trans.map((t, i) => Math.abs(t[0] - sp.trans[i][0]))) : NaN;
  let dTurn = 0, dTh = 0, dRho = 0;
  for (let i = 0; i < L3.length; i++) { dTurn = Math.max(dTurn, Math.abs(L3[i].turn - Lp[i].turn)); dTh = Math.max(dTh, circ(L3[i].th, Lp[i].th)); dRho = Math.max(dRho, Math.abs(L3[i].rho - Lp[i].rho)); }
  const ok = trig && seq && s3.drops.length === sp.drops.length && dTurn <= 0.005 && dTh <= 0.01 && dRho <= 0.01;
  allOk &&= ok;
  console.log(`${tr}: ${ok ? 'PASS' : 'FAIL'} | shifts ${sp.shifts} drops ${sp.drops} (nav3 ${s3.drops}) rearms ${sp.rearms} | triggers exact ${trig} | transitions ${sp.trans.length} (nav3 ${s3.trans.length}) seq ${seq} max |Δf| ${maxDt} | max |Δturn| ${dTurn.toExponential(2)} |Δth| ${dTh.toExponential(2)} |Δrho| ${dRho.toExponential(2)}`);
}
console.log(allOk ? '6b: all PASS' : '6b: FAIL');
